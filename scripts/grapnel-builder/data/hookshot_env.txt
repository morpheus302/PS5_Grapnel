# Grapnel on the PS5: the settings the game reads at start, from its data folder: /data/grapnel/hookshot_env.txt (the
# file keeps the name hookshot_env.txt). One NAME=value per line; a line starting with # is a comment. These are the
# settings the port is played and tested with on a base PS5; change one only if you know what it does.
# A Hookshot port's folder from before, /data/hookshot, is renamed to /data/grapnel at the first start (nothing copied).

# ---- the renderer's memory: the console has about 12 GiB shared by the CPU and the GPU ----
# no VRAM mirrors (the console's memory is shared); the texture budget
RX_VK_MIRROR=0
RX_VK_VRAM_MB=1024
# the driver's emulated task draws, in chunks of this many workgroups
RADV_PS5_TASK_WORKGROUPS=65536
# no pipeline precompiling at start; the driver's own shader cache (in the title folder) may grow to 4 GiB
RX_VK_PRECOMPILE=0
MESA_SHADER_CACHE_MAX_SIZE=4G
# guest memory backed in 64 KiB blocks at its first touch; imports rounded to 4 MiB; 1 MiB thread stacks
HOOKSHOT_PS5_FAULT_KB=64
RX_VK_IMPORT_CHUNK_MB=4
HOOKSHOT_PS5_STACK_MB=1
# the GPU programs listed in skip.txt (one key a line) are left out: a program whose compile on the console takes seconds
# and about 500 MiB (Lost Woods). Ship skip.txt beside this file in the data folder
RX_VK_SKIP_FILE=/data/grapnel/skip.txt
# decoded ASTC textures: kept as BC3, large ones without their top levels once they would take 4 MB
RX_VK_TEX_MAX_DIM=512
RX_VK_ASTC_BC_MB=0
RX_VK_TEX_SKIP_MB=4
# the pipeline file of the PC build is neither read nor written: the driver's own cache serves
RX_VK_PCACHE_FILE=0
# the title closes itself, with a line in its log, when less than 192 MiB of direct memory is left (instead of letting the
# console run out)
HOOKSHOT_PS5_MEMORY_FLOOR_MB=192
# mesh and task programs as compute shaders (the console's mesh shaders lose the GPU)
RX_VK_FORCE_MESH_FALLBACK=1
# smaller allocation blocks, two background workers, smaller upload arenas, a bounded render target pool
RX_VK_VMA_BLOCK_MB=16
RX_VK_IMAGE_BLOCK_MB=4
RX_VK_WORKERS=2
RX_VK_STAGING_ARENA_MB=16
RX_VK_TARGET_POOL_MB=128
RX_VK_IMAGE_DEDICATED_KB=256
# the TV mode 2560x1440 (fewer output buffers than 3840x2160; the game draws 2048x1152 either way)
HOOKSHOT_PS5_DISPLAY=2560x1440
# textures nothing wrote are uploaded, not cleared on the GPU (the clear stopped the GPU)
RX_VK_ZERO_UPLOAD=0
# smaller descriptor pools
RX_VK_DESC_POOL_DIV=8
# the render target pool kept while the driver reports VRAM as short (always on the console)
RX_VK_TARGET_POOL_KEEP=1

# ---- GPU stability ----
# the driver waits for the compute engine after each dispatch, one recording a submission, no late parameter-cache
# allocation for vertex waves: the combination that runs without GPU hangs on this console
RADV_PS5_SYNC_CS=1
RX_VK_BATCH_MAX=1
RADV_PS5_LATE_ALLOC=off

# ---- memory given back during play ----
RX_VK_STAGING_KEEP_MB=64
RX_VK_CAP_SPARE_MB=32
HOOKSHOT_PS5_ZERO_RECLAIM=1
HOOKSHOT_PS5_HEAP_PIECES_KB=1024
RX_VK_EVICT_FREE_NOW=1
RX_VK_STAGING_RECHECK=1
RX_VK_EVICT_BURST_UNUSED=20
RX_VK_EVICT_BURST_MADE=20
# textures kept while at least 1200 MiB of direct memory stays free (fewer re-uploads in crowded places)
RX_VK_EVICT_KEEP_FREE_MB=1200
# dead render targets let go at every 2nd present
RX_VK_DEAD_EVERY=2
# cold memory: game memory unused for a while is written to the SSD (in the data folder, hookshot-cold-*.bin) and read back
# when touched again; at most 1024 MiB written a session. It lets the heaviest places (Castle Town) fit in memory
HOOKSHOT_PS5_COLD=1
HOOKSHOT_PS5_COLD_CAP_MB=1024

# ---- the picture ----
# the softer picture: the game's over-sharpened upscaling network replaced by the renderer's own, with FXAA
RX_VK_TAAU_KEYS=fd029a317a9a4b94,9dc98871eb7fc101
# characters' shadow maps sized in steps of 384 (the render targets after them stop moving every frame)
HOS_SHADOW_BUCKET=384
# the pause menu's 3D Link: no forced anisotropy on that one program's single-level image (the dark dots)
RX_VK_RT_NO_ANISO=becef27180cc822f

# ---- frame rate ----
# the render thread: the replay of the game's commands on a thread of its own
RX_VK_ASYNC=1
# the game's threads without virtual cores
HOS_VCORES=0
# clocks read from the processor's time stamp counter instead of a system call each time
RX_VK_CHEAP_CLOCK=1
RX_VK_REPORT_TSC=1
HOS_TSC_CLOCK=1
# a texture converted from ASTC to BC3 in one GPU pass for all its levels
RX_VK_BC_ONE_PASS=1
# the render thread's helper threads off its CPU
HOOKSHOT_PS5_HELPERS_SHARED=1
# a stage's textures resolved in one call (lighter render thread)
RX_VK_TEX_LEAN=1

# ---- the console's menus ----
# the DualSense's button icons in the game (made on the PC from your own game files, in the data folder's prompts/;
# without them the game keeps its own icons)
HOOKSHOT_PS5_PROMPTS=1
# the in-game menu (touchpad + R1) offers "Back to the Grapnel screen" (the launch screen)
HOOKSHOT_PS5_BACK_TO_LAUNCHER=1
# the in-game menu shows the console's settings only
HOOKSHOT_PS5_MENU_CONSOLE=1

# ---- the shaders ----
# the builder prepares the game's shaders on the PC (its step 7): the console's driver finds them in its cache, the title folder's
# radv-shader-cache, and the launch screen only checks it. HOOKSHOT_PS5_PREPARE=1 would let the console prepare them itself from
# its SHADERS page (hours on a base PS5, and the census beside the game): a fallback for development, off
